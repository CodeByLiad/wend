cmake_minimum_required(VERSION 3.22.1)
project(wend_llama LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Native inference is always optimized for benchmarking" FORCE)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama
    URL https://github.com/ggml-org/llama.cpp/archive/93cf736a8d1d46aa922e27eb64aa712bcd5ebc30.tar.gz
    URL_HASH SHA256=92c2fa7772ab215a7baec7346fdd1581c77de2008ac56fb64327adbf122db8a2)
FetchContent_MakeAvailable(llama)
add_library(wend_llama SHARED benchmark.cpp)
target_compile_features(wend_llama PRIVATE cxx_std_17)
target_compile_options(wend_llama PRIVATE -O3)
target_link_libraries(wend_llama PRIVATE llama android log)
